% ============================================================
%  A1 - Befehle mit Logik (Musterlösung)
%  Kompilieren: pdflatex uebungsblatt.tex
% ============================================================
\documentclass[a4paper,11pt]{article}
\usepackage[T1]{fontenc}
\usepackage[ngerman]{babel}
\usepackage{pgffor}          % stellt \foreach bereit (Schleifen)

% ------------------------------------------------------------
% Schritt 1: Befehl mit Sternvariante
%   s  = optionaler Stern, abfragbar mit \IfBooleanTF{#1}{mit Stern}{ohne Stern}
% ------------------------------------------------------------
\NewDocumentCommand{\fh}{s}{%
  \IfBooleanTF{#1}%
    {University of Applied Sciences Upper Austria}% \fh*
    {FH Oberösterreich}%                            \fh
}

% ------------------------------------------------------------
% Schritt 2: Zähler
%   aufgabe = laufende Nummer, punkte = Summe aller Punkte
% ------------------------------------------------------------
\newcounter{aufgabe}
\newcounter{punkte}

% ------------------------------------------------------------
% Schritt 3: \aufgabe[Punkte]{Titel}
%   O{0} = optionales Argument in [ ], Standardwert 0
%   m    = Pflichtargument in { }
% ------------------------------------------------------------
\NewDocumentCommand{\aufgabe}{O{0} m}{%
  \stepcounter{aufgabe}%            Nummer um 1 erhöhen
  \addtocounter{punkte}{#1}%        Punkte zur Summe addieren
  \subsection*{Aufgabe \theaufgabe: #2%
    \ifnum#1>0 %                    Punkte nur anzeigen, wenn welche vergeben sind
      \hfill{\normalfont\small(#1~\ifnum#1=1 Punkt\else Punkte\fi)}%
    \fi}%
}

% ------------------------------------------------------------
% Schritt 4: Schleife - \antwortzeilen{n} zeichnet n Schreiblinien
% ------------------------------------------------------------
\NewDocumentCommand{\antwortzeilen}{m}{%
  \par
  \foreach \i in {1,...,#1}{%
    \noindent\rule{0pt}{8mm}\rule{\linewidth}{0.4pt}\par
  }%
}

\begin{document}

\section*{Übungsblatt}
Erstellt an der \fh{} (englisch: \fh*).

\aufgabe[3]{Dokumentklassen}
Nennen Sie drei Standardklassen von \LaTeX.
\antwortzeilen{3}

\aufgabe[1]{Kompilieren}
Welches Programm erzeugt aus einer \texttt{.tex}-Datei ein PDF?
\antwortzeilen{1}

\aufgabe{Feedback}
Was hat Ihnen am Workshop gefallen? (Ohne Punkte)
\antwortzeilen{2}

\bigskip
\noindent\textbf{Erreichbare Punkte: \thepunkte}

\end{document}
